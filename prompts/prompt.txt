### Step 3: Example Code Implementation (React / Next.js example)

Create a component at `pages/auth/verification-approved.jsx` (or `src/app/auth/verification-approved/page.tsx`):

```jsx
import { useEffect, useState } from 'react';
import { useRouter, useSearchParams } from 'next/navigation'; // Or react-router-dom `useSearchParams`

export default function VerificationApprovedPage() {
  const router = useRouter();
  const searchParams = useSearchParams();

  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const userId = searchParams.get('userId');
    const role = searchParams.get('role');
    const status = searchParams.get('status');
    const token = searchParams.get('token');

    if (!token || status !== 'accepted') {
      setError('Invalid or expired verification link.');
      setLoading(false);
      return;
    }

    async function handleVerification() {
      try {
        // 1. (Optional) Validate token with backend if required
        const res = await fetch('/api/v1/deep-links/verify', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ data: userId, token })
        });

        // 2. Save notice or user state if logged in
        localStorage.setItem('clinic_approval_status', 'accepted');

        // 3. Redirect user to Login or Subscriptions page
        const isLoggedIn = Boolean(localStorage.getItem('token')); // check current auth state

        if (!isLoggedIn) {
          // Redirect to login page with return URL pointing to subscriptions
          router.push('/login?redirect=/subscriptions&approved=true');
        } else {
          // User is already logged in -> Go straight to Subscriptions
          router.push('/subscriptions');
        }
      } catch (err) {
        setError('Verification failed. Please try again or contact support.');
      } finally {
        setLoading(false);
      }
    }

    handleVerification();
  }, [searchParams, router]);

  if (loading) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen">
        <h2 className="text-xl font-semibold">Verifying your clinic account...</h2>
        <p className="text-gray-500">Please wait while we redirect you.</p>
      </div>
    );
  }

  if (error) {
    return (
      <div className="flex flex-col items-center justify-center min-h-screen">
        <h2 className="text-xl font-bold text-red-600">Verification Error</h2>
        <p className="text-gray-600">{error}</p>
        <button 
          onClick={() => router.push('/login')}
          className="mt-4 px-4 py-2 bg-blue-600 text-white rounded"
        >
          Go to Login
        </button>
      </div>
    );
  }

  return null;
}
```
